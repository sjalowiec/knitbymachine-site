function matchesQuery(wizard: Wizard, query: string): boolean {
  if (!query) return true;
  const q = query.toLowerCase();

  const inTitle = wizard.title.toLowerCase().includes(q);
  const inCategory = wizard.category?.toLowerCase().includes(q) ?? false;

  const inKeywords =
    Array.isArray(wizard.keywords) &&
    wizard.keywords.some((kw) => kw.toLowerCase().includes(q));

  const inAliases =
    Array.isArray(wizard.aliases) &&
    wizard.aliases.some((a) => a.toLowerCase().includes(q));

  return inTitle || inCategory || inKeywords || inAliases;
}
