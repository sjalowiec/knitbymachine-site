4. Add the modal open/close JavaScript

Right after your existing <script> block (where init() is called), but before </body>, add this small script:

    <script>
      (function () {
        const openBtn = document.getElementById("openAskModal");
        const modal = document.getElementById("askModal");
        const closeBtn = document.getElementById("closeAskModal");
        const overlay = modal?.querySelector(".ask-modal__overlay");

        if (!openBtn || !modal) return;

        function openModal(e) {
          if (e) e.preventDefault();
          modal.removeAttribute("hidden");
          document.body.style.overflow = "hidden";
        }

        function closeModal() {
          modal.setAttribute("hidden", "true");
          document.body.style.overflow = "";
        }

        openBtn.addEventListener("click", openModal);
        closeBtn?.addEventListener("click", closeModal);
        overlay?.addEventListener("click", closeModal);

        // Optional: close on Escape key
        document.addEventListener("keydown", (e) => {
          if (e.key === "Escape" && !modal.hasAttribute("hidden")) {
            closeModal();
          }
        });
      })();
    </script>


Make sure this new <script> comes after your big ANA script, not inside it.
