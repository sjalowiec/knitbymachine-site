export async function waitForAuth({ timeoutMs = 8000, intervalMs = 150 } = {}) {
  const start = Date.now();

  const getUser = () => {
    // Future: replace with Memberstack / Outseta user
    if (window.KBM_AUTH?.user) return window.KBM_AUTH.user;

    // Dev fallback
    try {
      const raw = localStorage.getItem("kbm_user");
      if (raw) return JSON.parse(raw);
    } catch (e) {}

    return null;
  };

  return new Promise((resolve) => {
    const tick = () => {
      const user = getUser();
      if (user) return resolve({ ok: true, user });
      if (Date.now() - start > timeoutMs) return resolve({ ok: false, user: null });
      setTimeout(tick, intervalMs);
    };
    tick();
  });
}

export function setHyvorIdentity({ websiteId, user }) {
  window.hyvorTalk = window.hyvorTalk || {};
  window.hyvorTalk.website = websiteId;

  window.hyvorTalk.sso = {
    user: {
      id: String(user.id || user.userId || user.email || "unknown"),
      name: String(user.name || user.firstName || "Member"),
      email: String(user.email || ""),
      avatar: user.avatar || ""
    }
  };
}

export function loadHyvorScript() {
  if (document.querySelector('script[data-kbm-hyvor="1"]')) return;

  const s = document.createElement("script");
  s.async = true;
  s.src = "https://talk.hyvor.com/embed/embed.js";
  s.dataset.kbmHyvor = "1";
  document.body.appendChild(s);
}

4b. Add this client-side loader script
<script type="module">
  import { waitForAuth, setHyvorIdentity, loadHyvorScript } from "/src/lib/hyvorLoader.js";

  const HYVOR_WEBSITE_ID = "14706";

  const mount = document.getElementById("hyvor-talk-view");
  if (mount) {
    mount.insertAdjacentHTML("beforebegin", `
      <div class="kbm-hyvor-wait" id="kbm-hyvor-wait">
        Loading your workshop conversation…
      </div>
    `);
  }

  const result = await waitForAuth({ timeoutMs: 8000 });

  if (result.ok && result.user) {
    const waitEl = document.getElementById("kbm-hyvor-wait");
    if (waitEl) waitEl.remove();

    setHyvorIdentity({ websiteId: HYVOR_WEBSITE_ID, user: result.user });
    loadHyvorScript();
  } else {
    // Do not load Hyvor if auth never appears
    const waitEl = document.getElementById("kbm-hyvor-wait");
    if (waitEl) {
      waitEl.innerHTML = "Loading is taking longer than expected. Refresh this page if needed.";
    }
  }
</script>
