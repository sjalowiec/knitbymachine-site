1) Update the <script> inside the Confidence Path section

In home-confidence-test.astro, find the <script> that targets #confidence-path and updates #confidence-description.

Replace the entire script block with this:

<script>
  const section = document.getElementById('confidence-path');
  if (section) {
    const pills = section.querySelectorAll('[data-description]');
    const desc = section.querySelector('#confidence-description');

    pills.forEach((pill) => {
      const update = () => {
        if (!desc) return;

        // change the text
        desc.textContent = pill.dataset.description || '';

        // retrigger animation
        desc.classList.remove('confidence-description-animate');
        // force reflow so the animation can restart
        void desc.offsetWidth;
        desc.classList.add('confidence-description-animate');
      };

      pill.addEventListener('mouseenter', update);
      pill.addEventListener('focus', update);
      pill.addEventListener('click', update);
    });
  }
</script>
