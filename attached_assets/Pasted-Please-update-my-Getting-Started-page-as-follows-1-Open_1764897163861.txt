6. Add JavaScript to open/close the modal with autoplay + ESC

Immediately after the modal HTML, add this script block:

<script>
  const modal = document.getElementById("welcome-video-modal");
  const iframe = document.getElementById("welcome-video-iframe");
  const openBtn = document.getElementById("open-welcome-video");
  const closeBtn = document.getElementById("close-welcome-video");

  const videoUrl =
    "https://player.vimeo.com/video/1143645774?autoplay=1&title=0&byline=0&portrait=0";

  function openModal() {
    if (!modal || !iframe) return;
    modal.style.display = "flex";
    iframe.src = videoUrl;
  }

  function closeModal() {
    if (!modal || !iframe) return;
    modal.style.display = "none";
    iframe.src = "";
  }

  if (openBtn) {
    openBtn.addEventListener("click", openModal);
  }

  if (closeBtn) {
    closeBtn.addEventListener("click", closeModal);
  }

  if (modal) {
    // close when clicking the dark backdrop
    modal.addEventListener("click", (e) => {
      if (e.target === modal) {
        closeModal();
      }
    });
  }

  // close on ESC key
  document.addEventListener("keydown", (e) => {
    if (e.key === "Escape" && modal && modal.style.display === "flex") {
      closeModal();
    }
  });
</script>


Behavior:

Clicking the thumbnail/button opens the modal and autoplays the Vimeo video

Clicking the ✕ button closes the modal and stops the video (by clearing src)

Clicking outside the modal (on the dark backdrop) closes it

Pressing Esc closes it
