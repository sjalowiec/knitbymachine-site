Step 3 – Add the JavaScript that powers the flipbook

At the bottom of this Skill Builder detail page/component, add a <script> block with is:inline (so Astro will include it as raw JS). This script should:

Read all .kbm-step-card elements into an array.

Dynamically populate the .kbm-stepper-progress list (<ol>) with one <li> per step.

Handle:

Clicking Previous / Next buttons

Clicking progress items

Updating the current step number (#kbm-current-step)

Updating the total (#kbm-total-steps)

Updating the current step title (#kbm-current-step-title)

Showing/hiding cards via the kbm-step-card--active class

Enabling/disabling arrows at first and last step

Use this logic (you can adapt variable names, but keep behavior the same):

<script is:inline>
  (function () {
    const stepCards = Array.from(document.querySelectorAll('.kbm-step-card'));
    if (!stepCards.length) return;

    const prevBtn = document.querySelector('.kbm-stepper-nav--prev');
    const nextBtn = document.querySelector('.kbm-stepper-nav--next');
    const currentStepEl = document.getElementById('kbm-current-step');
    const totalStepsEl = document.getElementById('kbm-total-steps');
    const currentTitleEl = document.getElementById('kbm-current-step-title');
    const progressList = document.querySelector('.kbm-stepper-progress');

    const totalSteps = stepCards.length;
    let currentStep = 1;

    if (totalStepsEl) totalStepsEl.textContent = String(totalSteps);

    // Build progress items dynamically so they match the number of steps
    if (progressList) {
      progressList.innerHTML = '';
      stepCards.forEach((card, index) => {
        const step = index + 1;
        const li = document.createElement('li');
        li.className = 'kbm-stepper-progress__item' + (step === 1 ? ' kbm-stepper-progress__item--active' : '');
        li.dataset.step = String(step);
        li.textContent = String(step);
        li.addEventListener('click', () => showStep(step));
        progressList.appendChild(li);
      });
    }

    const progressItems = Array.from(document.querySelectorAll('.kbm-stepper-progress__item'));

    function showStep(step) {
      currentStep = step;

      stepCards.forEach(card => {
        const cardStep = Number(card.getAttribute('data-step'));
        card.classList.toggle('kbm-step-card--active', cardStep === currentStep);
      });

      progressItems.forEach(item => {
        const itemStep = Number(item.getAttribute('data-step'));
        item.classList.toggle('kbm-stepper-progress__item--active', itemStep === currentStep);
      });

      if (currentStepEl) currentStepEl.textContent = String(currentStep);

      const activeCard = stepCards.find(card => Number(card.getAttribute('data-step')) === currentStep);
      if (activeCard && currentTitleEl) {
        currentTitleEl.textContent = activeCard.getAttribute('data-step-title') || '';
      }

      if (prevBtn) prevBtn.disabled = currentStep === 1;
      if (nextBtn) nextBtn.disabled = currentStep === totalSteps;
    }

    function handleNavClick(evt) {
      const direction = evt.currentTarget.getAttribute('data-direction');
      if (direction === 'prev' && currentStep > 1) {
        showStep(currentStep - 1);
      }
      if (direction === 'next' && currentStep < totalSteps) {
        showStep(currentStep + 1);
      }
    }

    if (prevBtn) prevBtn.addEventListener('click', handleNavClick);
    if (nextBtn) nextBtn.addEventListener('click', handleNavClick);

    // Optional: keyboard arrow navigation
    document.addEventListener('keydown', (evt) => {
      if (evt.key === 'ArrowRight' && currentStep < totalSteps) {
        showStep(currentStep + 1);
      } else if (evt.key === 'ArrowLeft' && currentStep > 1) {
        showStep(currentStep - 1);
      }
    });

    // Initialize
    showStep(1);
  })();
</script>
