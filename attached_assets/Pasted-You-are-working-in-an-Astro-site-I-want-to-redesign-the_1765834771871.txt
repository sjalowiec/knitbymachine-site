Suggested markup (adapt to actual variable names)

Use something like:

<ul class="plan-grid" aria-label="Workshop plan">
  {days.map((d) => (
    <li class="plan-item">
      <a class="plan-card" href={d.href}>
        <div class="plan-card-top">
          <span class="plan-day">Day {d.dayNumber}</span>
          {d.dayNumber === 1 && <span class="plan-badge">Start here</span>}
        </div>

        <h3 class="plan-title">{d.title}</h3>

        {d.description && <p class="plan-desc">{d.description}</p>}

        {d.duration && (
          <div class="plan-meta">
            <span class="plan-time">{d.duration}</span>
            <span class="plan-arrow" aria-hidden="true">→</span>
          </div>
        )}
        {!d.duration && (
          <div class="plan-meta">
            <span class="plan-arrow" aria-hidden="true">→</span>
          </div>
        )}
      </a>
    </li>
  ))}
</ul>
