Add a tiny client-side enhancement that swaps in Outseta name if available.
Right after the existing welcome dismiss script (or in the same script block area), add a small script that:

Looks for a user name from Outseta (try common locations without throwing errors)

If found, replaces {name} in the title text

If not found, leaves the server-rendered fallback name alone

Add this in the page (module script is fine):

<script type="module">
  (function () {
    const titleEl = document.querySelector('.gw-welcome-title');
    if (!titleEl) return;

    // If already dismissed, don't waste time
    const dismissed = localStorage.getItem('kbm_gw_welcome_dismissed') === '1';
    if (dismissed) return;

    // Try to pull a name from Outseta (non-fatal if not present)
    const tryGetName = () => {
      try {
        // Common possibilities depending on Outseta setup
        const o =
          window.Outseta ||
          window.outseta ||
          window.os ||
          null;

        // Try a few plausible shapes without assuming too much
        const fromPossibleObject =
          o?.user?.FullName ||
          o?.user?.fullName ||
          o?.user?.Name ||
          o?.user?.name ||
          o?.auth?.user?.FullName ||
          o?.auth?.user?.fullName ||
          o?.auth?.user?.name ||
          null;

        if (typeof fromPossibleObject === 'string' && fromPossibleObject.trim()) {
          return fromPossibleObject.trim();
        }

        // As a final fallback, sometimes user info is stashed in localStorage by auth layers
        const ls =
          localStorage.getItem('outseta_user') ||
          localStorage.getItem('OutsetaUser') ||
          null;

        if (ls) {
          const parsed = JSON.parse(ls);
          const n = parsed?.FullName || parsed?.fullName || parsed?.name;
          if (typeof n === 'string' && n.trim()) return n.trim();
        }
      } catch (e) {
        // ignore
      }
      return null;
    };

    const name = tryGetName();
    if (!name) return;

    // Replace only if template style exists, otherwise append politely
    const current = titleEl.textContent || '';
    if (current.includes('{name}')) {
      titleEl.textContent = current.replace('{name}', name);
    } else if (current.toLowerCase().includes('welcome')) {
      // If already rendered (server used fallback), swap the part after comma if it looks like "Welcome, X"
      titleEl.textContent = current.replace(/^(Welcome,\s*)(.+)$/i, `$1${name}`);
    }
  })();
</script>


Important:

This does NOT require Outseta to be fully wired today.

It will “just start working” later when Outseta user data exists on the page.
