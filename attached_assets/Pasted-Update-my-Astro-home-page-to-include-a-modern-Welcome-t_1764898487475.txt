3. Add the open/close behavior with vanilla JS
In the same Astro home page file where you inserted the HTML, add a small inline script at the bottom of the file (before the closing </body> if it exists, or inside the Astro template near the end). Use is:inline so it runs client-side:
<script is:inline>
  document.addEventListener('DOMContentLoaded', () => {
    const wrapper = document.querySelector('[data-kbm-intro-wrapper]');
    const banner = document.querySelector('[data-kbm-intro-banner]');
    const drawer = document.querySelector('[data-kbm-intro-drawer]');
    const openButtons = document.querySelectorAll('[data-kbm-intro-open]');
    const closeButtons = document.querySelectorAll('[data-kbm-intro-close]');
    const dismissButtons = document.querySelectorAll('[data-kbm-intro-dismiss]');

    if (!wrapper || !banner || !drawer) return;

    const introDismissed = window.localStorage.getItem('kbmIntroDismissed') === 'true';

    // If the user has previously dismissed it, hide the whole thing
    if (introDismissed) {
      wrapper.classList.add('hidden');
      return;
    }

    const openDrawer = () => {
      drawer.classList.add('kbm-intro-drawer--open');
      drawer.setAttribute('aria-hidden', 'false');
    };

    const closeDrawer = () => {
      drawer.classList.remove('kbm-intro-drawer--open');
      drawer.setAttribute('aria-hidden', 'true');
    };

    openButtons.forEach((btn) => {
      btn.addEventListener('click', () => {
        openDrawer();
      });
    });

    closeButtons.forEach((btn) => {
      btn.addEventListener('click', () => {
        // Close the drawer but keep the banner visible so they can reopen it
        closeDrawer();
      });
    });

    dismissButtons.forEach((btn) => {
      btn.addEventListener('click', () => {
        // User doesn’t want to see this again
        window.localStorage.setItem('kbmIntroDismissed', 'true');
        wrapper.classList.add('hidden');
      });
    });
  });
</script>

4. Overall behavior


On first visit, the banner appears under the main header with the “Show me around” button.


Clicking “Show me around” smoothly slides down the top drawer with the short intro, copy, and buttons.


Clicking the “×” in the drawer closes the drawer, but leaves the banner so they can open it again.


Clicking “Skip for now” hides the entire welcome UI and sets localStorage.kbmIntroDismissed = "true", so the banner/drawer won’t show on future visits (for the next few months while I leave this in place).
