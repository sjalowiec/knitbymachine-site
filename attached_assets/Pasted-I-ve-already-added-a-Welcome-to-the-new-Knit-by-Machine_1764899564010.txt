3. Update the JavaScript behavior (inline link open, close hides permanently)

In the same home page Astro file, update the inline script that handles the intro drawer so it matches this behavior:

Clicking “see a quick intro” opens the drawer.

Clicking the drawer × closes the drawer but keeps the banner.

Clicking the banner × closes the entire wrapper and sets localStorage.kbmIntroDismissed = "true" so the banner + drawer stay hidden on future visits.

Replace the existing script with this:

<script is:inline>
  document.addEventListener('DOMContentLoaded', () => {
    const wrapper = document.querySelector('[data-kbm-intro-wrapper]');
    const banner = document.querySelector('[data-kbm-intro-banner]');
    const drawer = document.querySelector('[data-kbm-intro-drawer]');
    const openButtons = document.querySelectorAll('[data-kbm-intro-open]');
    const drawerCloseButtons = document.querySelectorAll('[data-kbm-intro-close]');
    const bannerDismissButtons = document.querySelectorAll('[data-kbm-intro-dismiss]');

    if (!wrapper || !banner || !drawer) return;

    const introDismissed = window.localStorage.getItem('kbmIntroDismissed') === 'true';

    if (introDismissed) {
      wrapper.classList.add('hidden');
      return;
    }

    const openDrawer = () => {
      drawer.classList.add('kbm-intro-drawer--open');
      drawer.setAttribute('aria-hidden', 'false');
    };

    const closeDrawer = () => {
      drawer.classList.remove('kbm-intro-drawer--open');
      drawer.setAttribute('aria-hidden', 'true');
    };

    openButtons.forEach((btn) => {
      btn.addEventListener('click', () => {
        openDrawer();
      });
    });

    drawerCloseButtons.forEach((btn) => {
      btn.addEventListener('click', () => {
        // Close only the drawer, keep the banner visible
        closeDrawer();
      });
    });

    bannerDismissButtons.forEach((btn) => {
      btn.addEventListener('click', () => {
        // Hide banner + drawer permanently for this browser
        window.localStorage.setItem('kbmIntroDismissed', 'true');
        wrapper.classList.add('hidden');
      });
    });
  });
</script>
