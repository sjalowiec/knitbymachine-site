PROJECT TYPE: Astro project (Knit by Machine)

GOAL:
Create a reusable tooltip component using a hexagon outline with a “?” inside. The tooltip appears on hover/focus and displays a short text snippet, styled with KBM colors.

FILES TO CREATE
1. Create this file:

src/components/KBMTooltip.astro

FILE CONTENTS:

---
interface Props {
  text: string;
  position?: "top" | "right" | "bottom" | "left";
}

const { text, position = "top" } = Astro.props;
const positionClass = `kbm-tooltip--${position}`;
---

<span class={`kbm-tooltip ${positionClass}`} tabindex="0">
  <svg
    class="kbm-tooltip-icon"
    viewBox="0 0 100 100"
    width="18"
    height="18"
    aria-hidden="true"
  >
    <polygon
      points="50 2, 95 25, 95 75, 50 98, 5 75, 5 25"
      fill="none"
      stroke="#52682D"
      stroke-width="6"
      stroke-linejoin="round"
    />
    <text
      x="50"
      y="63"
      text-anchor="middle"
      font-size="55"
      font-family="Arial, sans-serif"
      fill="#52682D"
      pointer-events="none"
    >
      ?
    </text>
  </svg>

  <span class="kbm-tooltip-text" role="tooltip">
    {text}
  </span>
</span>
