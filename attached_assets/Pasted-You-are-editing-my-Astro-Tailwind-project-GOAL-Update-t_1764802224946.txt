<script>
  // -------------------------------------------
  // Confetti fallback (if no existing confetti lib)
  function triggerConfettiFallback() {
    document.body.animate(
      [
        { transform: 'translateY(0px)' },
        { transform: 'translateY(-10px)' },
        { transform: 'translateY(0px)' },
      ],
      { duration: 300, iterations: 1 }
    );
  }

  // -------------------------------------------
  // Button click handler: confetti + scroll
  function handleGettingStartedComplete() {
    // Confetti
    if (typeof window.triggerConfetti === "function") {
      window.triggerConfetti();
    } else {
      triggerConfettiFallback();
    }

    // Smooth scroll to the “What’s Next?” section
    const section = document.getElementById("whats-next");
    if (section) {
      section.scrollIntoView({ behavior: "smooth", block: "start" });
    }
  }
  window.handleGettingStartedComplete = handleGettingStartedComplete;

  // -------------------------------------------
  // Show the hidden button
  window.showGettingStartedCompleteButton = function () {
    const btn = document.getElementById("getting-started-complete-btn");
    if (btn) {
      console.log("[KBM] Showing Getting Started complete button");
      btn.classList.remove("hidden");
    } else {
      console.warn("[KBM] Button with id 'getting-started-complete-btn' not found");
    }
  };

  // -------------------------------------------
  // MindPal completion hook with logging
  window.addEventListener("message", (event) => {
    console.log("[KBM] window message received:", event.data);

    const data = event.data;

    const isMindPalCompleted =
      data === "mindpal-chat-completed" ||
      (data &&
        typeof data === "object" &&
        (
          data.type === "mindpal-chat-completed" ||
          data.event === "mindpal-chat-completed" ||
          data.action === "mindpal-chat-completed"
        ));

    if (isMindPalCompleted) {
      console.log("[KBM] MindPal reported chat completed – revealing button");
      window.showGettingStartedCompleteButton?.();
    }
  });
</script>
