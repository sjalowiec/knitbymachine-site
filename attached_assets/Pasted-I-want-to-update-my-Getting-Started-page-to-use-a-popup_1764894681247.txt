4. Add the JavaScript to open/close the popup and autoplay the video

In the same Astro file, add this <script> block at the bottom of the page (after the modal markup). Use inline JS:

<script>
  if (typeof window !== "undefined") {
    const openBtn = document.getElementById("open-welcome-video");
    const modal = document.getElementById("welcome-video-modal");
    const closeBtn = document.getElementById("close-welcome-video");
    const iframe = document.getElementById("welcome-video-iframe");

    const videoUrl =
      "https://player.vimeo.com/video/1143645774?autoplay=1&title=0&byline=0&portrait=0";

    function openModal() {
      if (!modal || !iframe) return;
      modal.classList.remove("hidden");
      modal.classList.add("flex");
      // set src when opened so autoplay works
      iframe.src = videoUrl;
    }

    function closeModal() {
      if (!modal || !iframe) return;
      modal.classList.add("hidden");
      modal.classList.remove("flex");
      // clear src to stop playback when closed
      iframe.src = "";
    }

    if (openBtn) {
      openBtn.addEventListener("click", openModal);
    }

    if (closeBtn) {
      closeBtn.addEventListener("click", closeModal);
    }

    if (modal) {
      // close when clicking the dark background
      modal.addEventListener("click", (event) => {
        if (event.target === modal) {
          closeModal();
        }
      });
    }
  }
</script>


Important:

The autoplay=1 parameter in the Vimeo URL should cause the video to autoplay once the modal opens, since it’s triggered by a user click on the button.

Clearing iframe.src on close stops the video audio when the modal is dismissed.
